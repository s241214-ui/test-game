import tkinter as tk import math import time class Fake3DGame: def __init__(self, root): self.root = root self.root.title("Tkinter Pseudo-3D FPS") self.canvas = tk.Canvas(root, width=1000, height=700, bg='lightblue') self.canvas.pack() # 游戏状态变量 self.player_x = 0 self.player_y = 0 self.player_z = 0 # 跳跃高度 self.is_locked = False # Fixed Action 状态 self.held_item = None # 当前拿着的物品 # 摇杆数据 (模拟值，范围 -1.0 到 1.0) self.left_stick = {'x': 0, 'y': 0} self.right_stick = {'x': 0, 'y': 0} # 动画状态 self.weapon_angle = 0 self.is_swinging = False # 绑定鼠标拖动来模拟摇杆 (实际游戏中可以用虚拟摇杆) self.canvas.bind("", lambda e: self.move_stick(e, 'left')) self.canvas.bind("", lambda e: self.reset_sticks(e, 'left')) self.canvas.bind("", lambda e: self.move_stick(e, 'right')) self.canvas.bind("", lambda e: self.reset_sticks(e, 'right')) # 键盘快捷键测试（比如空格攻击） self.root.bind("", self.start_swing) # 绘制 UI 和按钮 (这里只是简单示意) self.setup_ui() # 启动游戏循环 self.game_loop() def setup_ui(self): # 绘制左摇杆背景 self.canvas.create_oval(150, 500, 350, 700, outline='black', width=3) # 绘制右摇杆背景 self.canvas.create_oval(750, 500, 950, 700, outline='green', width=3) # 绘制按钮 (左侧) self.btn_jump = tk.Button(self.root, text="Jump", command=self.do_jump) self.canvas.create_window(150, 450, window=self.btn_jump) self.btn_fixed = tk.Button(self.root, text="Fixed Action", command=self.toggle_lock) self.canvas.create_window(250, 450, window=self.btn_fixed) self.btn_grab = tk.Button(self.root, text="Grab", command=self.do_grab) self.canvas.create_window(350, 450, window=self.btn_grab) # --- 按钮逻辑 --- def do_jump(self): if self.player_z == 0: # 只有在地上才能跳 self.player_z = 50 # 跳跃高度 # 这里可以加一个简单的下落动画 def toggle_lock(self): self.is_locked = not self.is_locked print(f"Fixed Action Active: {self.is_locked}") def do_grab(self): # 这里应该检查屏幕中心是否有物体，有的话捡起来 # 简单演示：直接拿起长矛 if self.held_item is None: self.held_item = "spear" print("Grabbed spear") def start_swing(self, event=None): if not self.is_swinging and self.held_item: self.is_swinging = True self.weapon_angle = 0 # 开始角度 # 动画由 game_loop 处理 # --- 摇杆逻辑 --- def move_stick(self, event, stick_type): if self.is_locked and stick_type == 'left': return # 如果锁定了，忽略左摇杆输入 # 简化的摇杆计算，计算拖拽中心点 # 实际应用中需要更精确地计算相对距离 print(f"{stick_type} stick moved to: {event.x}, {event.y}") def reset_sticks(self, event, stick_type): if stick_type == 'left': self.left_stick = {'x': 0, 'y': 0} else: self.right_stick = {'x': 0, 'y': 0} # --- 游戏循环与渲染 --- def game_loop(self): # 1. 处理移动逻辑 (根据 left_stick) if not self.is_locked: # self.player_x += self.left_stick['x'] * speed # self.player_y += self.left_stick['y'] * speed pass # 2. 处理跳跃物理 if self.player_z > 0: self.player_z -= 5 if self.player_z < 0: self.player_z = 0 # 3. 处理武器动画 if self.is_swinging: self.weapon_angle += 10 # 挥动速度 if self.weapon_angle > 90: # 挥动结束 self.weapon_angle = 0 self.is_swinging = False # 4. 重绘画面 self.draw_scene() # 5. 循环调用 self.root.after(30, self.game_loop) # 约 33 FPS def draw_scene(self): self.canvas.delete("all") # 清空画面 # 绘制背景 (随玩家位置偏移) # self.canvas.create_rectangle(...) # self.canvas.create_line(0, 350 + self.player_z, 1000, 350 + self.player_z, fill='green', width=2) # 地平线 # 绘制准星 self.canvas.create_oval(490, 340, 510, 360, outline='red') # 绘制手和武器 (右下角) if self.held_item == "spear": # 根据 weapon_angle 计算长矛的绘制坐标 # 这是一个简化的长矛绘制 start_x, start_y = 800, 600 end_x = start_x - 100 * math.cos(math.radians(self.weapon_angle)) end_y = start_y - 300 * math.sin(math.radians(self.weapon_angle)) self.canvas.create_line(start_x, start_y, end_x, end_y, fill='brown', width=5) # 绘制手 (简单的形状) self.canvas.create_oval(750, 550, 850, 650, fill='peachpuff', outline='black') # 绘制 UI 按钮 (如果删除了这里需要重新画，但我们在 setup_ui 里用了 Tkinter 组件，所以不需要画在 canvas 上) if __name__ == "__main__": root = tk.Tk() game = Fake3DGame(root) root.mainloop()
